# Packages installed with apt on Ubuntu 24.04 LTS. One per line; "#" a comment.
#
# Separate from ubuntu26/apt.txt because this tracks 24.04: versions differ
# throughout, and in one case enough to change where the package comes from.
# Homebrew formulae are not release-tied, so those are shared in
# ../common/brew.txt rather than duplicated here.
#
# Every entry below was verified present in a clean ubuntu:24.04 container.

# Homebrew prerequisites. build-essential is not optional: Homebrew on Linux
# compiles against the system toolchain.
build-essential
curl
file
git
procps

# System trust and signing. Wanted by apt itself, so brew's copies are no
# substitute.
ca-certificates
gnupg

# Shell and editors.
#
# neovim is NOT here, unlike on 26.04 -- see brew.txt. 24.04 ships 0.9.5 where
# 26.04 ships 0.11.6, and plugin configurations written against 0.10+ will not
# load on 0.9.
#
# tmux is 3.4 here against 3.6a on 26.04. The oh-my-tmux config in
# common/apps guards its newer settings behind %if version checks, so 3.4
# degrades rather than breaking, and apt's build stays matched to the system
# terminfo.
tmux
vim

# Remote access and transfer
openssh-server
rsync

# Network diagnostics. A minimal install has none of these -- not even ip, ss
# or nc, which fuller installs ship -- so every one is listed rather than
# assumed.
#
# From apt, not brew. ping and mtr work for an ordinary user only because the
# package grants them cap_net_raw when it is installed as root; the kernel's
# alternative, unprivileged ping sockets (net.ipv4.ping_group_range), is often
# switched off. A brew install runs as the user and cannot grant that, so its
# mtr would need sudo every time.
#
# Names are the ones that exist on both 24.04 and 26.04. 26.04 dropped
# dnsutils, so the real package bind9-dnsutils is listed; telnet is likewise
# only a transitional package for inetutils-telnet.
iputils-ping        # ping
traceroute          # traceroute
mtr-tiny            # mtr: ping and traceroute combined, live; no X11 deps
bind9-dnsutils      # dig, nslookup, host
whois               # whois
iproute2            # ip, ss
net-tools           # netstat, ifconfig, route, arp
netcat-openbsd      # nc
tcpdump             # packet capture; needs sudo to run
nmap                # host and port scanning
iperf3              # bandwidth between machines; its daemon is left off
socat               # relay data between sockets, poke at ports
inetutils-telnet    # telnet

# Utilities where 24.04 is at or near parity with brew: expect and unzip are
# identical builds, and jq, htop, make and tree are close enough that tracking
# the newer version buys nothing.
expect
htop
jq
make
tree
unzip

# psql and friends. brew's equivalent is the keg-only libpq, which has to be
# manually linked onto PATH; apt gives the client tools directly.
postgresql-client
