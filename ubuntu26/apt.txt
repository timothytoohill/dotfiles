# Packages installed with apt. One per line; "#" starts a comment.
#
# setup.sh installs only what is missing, so adding a line here and re-running
# does the minimum amount of work.
#
# What belongs here rather than in brew.txt: anything the system integrates
# with (trust stores, daemons, toolchains), and anything apt already ships at
# parity. Fast-moving CLI tools go in brew.txt, where they are not pinned to
# the distro release. See the README for the split.

# Homebrew prerequisites. Keep these: the brew stage needs them on a bare
# machine, and listing them here means one pass sets everything up.
# build-essential in particular is not optional -- Homebrew on Linux compiles
# against the system toolchain and refuses to work without it.
build-essential
curl
file
git
procps

# System trust and signing. Wanted by apt itself, so brew's copies are no
# substitute.
ca-certificates
gnupg

# Shell and editors
neovim
tmux
vim

# Remote access and transfer. Desktops get openssh-server as well, on
# purpose, so every machine can be reached the same way; on a laptop that
# means an SSH server on whatever network it joins.
openssh-server
rsync

# Network diagnostics. A minimal install has none of these -- not even ip, ss
# or nc, which fuller installs ship -- so every one is listed rather than
# assumed.
#
# From apt, not brew. ping and mtr work for an ordinary user only because the
# package grants them cap_net_raw when it is installed as root; the kernel's
# alternative, unprivileged ping sockets (net.ipv4.ping_group_range), is often
# switched off. A brew install runs as the user and cannot grant that, so its
# mtr would need sudo every time.
#
# Names are the ones that exist on both 24.04 and 26.04. 26.04 dropped
# dnsutils, so the real package bind9-dnsutils is listed; telnet is likewise
# only a transitional package for inetutils-telnet.
iputils-ping        # ping
traceroute          # traceroute
mtr-tiny            # mtr: ping and traceroute combined, live; no X11 deps
bind9-dnsutils      # dig, nslookup, host
whois               # whois
iproute2            # ip, ss
net-tools           # netstat, ifconfig, route, arp
netcat-openbsd      # nc
tcpdump             # packet capture; needs sudo to run
nmap                # host and port scanning
iperf3              # bandwidth between machines; its daemon is left off
socat               # relay data between sockets, poke at ports
inetutils-telnet    # telnet

# Utilities apt ships at parity with brew, or close enough that tracking the
# newer version buys nothing: unzip and make are identical builds, expect has
# not moved in years, and htop/tree/jq are within a patch release.
expect
htop
jq
make
tree
unzip

# psql and friends. brew's equivalent is the keg-only libpq, which has to be
# manually linked onto PATH; apt gives the client tools directly.
postgresql-client
